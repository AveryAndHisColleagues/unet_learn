class LightSSM(nn.Module):
    def __init__(self, dim):

        self.local = Conv3x3(DW) + Conv1x1 + BN + GELU

        self.mixer = SE-like gating:
            GAP → MLP → Sigmoid

        self.h_scan = Conv(1×3)
        self.w_scan = Conv(3×1)

        self.fuse = Conv1×1

    def forward(self, x):

        local = self.local(x)

        gate = self.mixer(local)
        gated = local * gate

        h = self.h_scan(gated)
        w = self.w_scan(gated)

        out = concat(local, h, w)
        out = self.fuse(out)

        return out + x
